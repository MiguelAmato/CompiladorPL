tipado(inst_do(lInstr1, Exp1, Exp2, lInstr2)) : 
	tipado(lInstr1)
	tipado(Exp1)
	tipado(Exp2)
	tipado(lInstr2)
	if (exp!(Exp1.tipo) == tipo_bool && exp!(Exp2.tipo) == tipo_bool)
		$.tipo = ambosOK(lInstr1, lInstr2)
	else
		if (exp!(Exp1.tipo) != tipo_bool)
			aviso_error(Exp1)
		endif
		if (exp!(Exp2.tipo) != tipo_bool)
			aviso_error(Exp2)
		endif
		$.tipo = error
	endif
