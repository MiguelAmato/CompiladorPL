		vincula(Instruccion0)

		vincula(Exp)
	else
		error

tipado(instruccion_alt(id, Instruccion0, Instruccion1, Exp)) : 
	tipado(id)
	if (ref!(id.tipo) != tipo_struct())
		$.tipo = error
	else
		tipado(Intruccion0)
		tipado(Instruccion1)
		$.tipo = ambasOk(Intruccion0.tipo, Instruccion1.tipo)
		tipado(Exp)
		if ($.tipo == ok  && Exp.tipo == tipo_bool)
			$.tipo = ok
		else 
			$.tipo = error
